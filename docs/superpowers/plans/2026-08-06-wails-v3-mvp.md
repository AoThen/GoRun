# Wails v3 版本实现计划（GoRun-wails）

> **面向 AI 代理的工作者：** 必需子技能：使用 superpowers:subagent-driven-development 或 superpowers:executing-plans 逐任务实现此计划。步骤使用复选框（`- [ ]`）语法来跟踪进度。

**目标：** 新增 `wails/` 目录，使用 **Go + Wails v3 + Vue 3 + TypeScript** 实现第三套完整 MVP，功能与 C++（`src/`）和 Rust（`rs/`）版本对齐：窗口框架、拖放添加、分类管理、项目管理、双击/管理员运行、全局快捷键、托盘图标、JSON 持久化。

**架构：** Wails v3 服务绑定模式 —— Go 侧提供 `ItemManager`/`Runner` 等服务（`application.Service`），前端 Vue 3 通过 `wails3 generate bindings -ts` 生成的类型安全 TS 绑定调用；Go 侧通过事件（`app.Event.Emit`）向前端推送拖放导入、运行结果等通知。

**技术栈：** Go 1.24 / Wails v3（`github.com/wailsapp/wails/v3`，beta 版）/ Vue 3 + Vite + TypeScript / 标准库 `encoding/json`（无额外存储依赖）/ WebView2 Runtime（Windows 10+ 自带）

**数据兼容：** JSON 结构与 `rs/src/model.rs`、`src/core/Types.h` 完全一致（camelCase 字段），`%EXEDIR%/config.json` 可直接互用。

```go
package app

import (
	"encoding/json"
	"os"
	"path/filepath"
)

type Storage struct {
	path string
	data AppData
}

func NewStorage(exeDir string) *Storage {
	return &Storage{path: filepath.Join(exeDir, "config.json")}
}

func (s *Storage) Load() error {
	b, err := os.ReadFile(s.path)
	if os.IsNotExist(err) {
		s.data = DefaultAppData()
		return nil
	}
	if err != nil {
		return err
	}
	s.data = DefaultAppData()
	if err := json.Unmarshal(b, &s.data); err != nil {
		return err
	}
	s.normalize()
	return nil
}

func (s *Storage) Save() error {
	b, err := json.MarshalIndent(s.data, "", "    ")
	if err != nil {
		return err
	}
	return os.WriteFile(s.path, b, 0644)
}

func (s *Storage) Data() *AppData { return &s.data }

func (s *Storage) normalize() {
	for i := range s.data.Categories {
		if s.data.Categories[i].IconSize == 0 {
			s.data.Categories[i].IconSize = 48
		}
	}
}
```

```go
package app

import (
	"errors"
	"os"
	"path/filepath"
	"sort"
)

type ItemManager struct {
	storage *Storage
}

func NewItemManager(s *Storage) *ItemManager { return &ItemManager{storage: s} }

func (m *ItemManager) GetData() AppData { return *m.storage.Data() }

func (m *ItemManager) AddCategory(name string) (Category, error) {
	if name == "" {
		return Category{}, errors.New("分类名称不能为空")
	}
	d := m.storage.Data()
	cat := Category{ID: GenerateID("cat"), Name: name, IconSize: 48}
	d.Categories = append(d.Categories, cat)
	return cat, m.storage.Save()
}

func (m *ItemManager) UpdateCategory(cat Category) error {
	d := m.storage.Data()
	for i := range d.Categories {
		if d.Categories[i].ID == cat.ID {
			d.Categories[i] = cat
			return m.storage.Save()
		}
	}
	return errors.New("分类不存在")
}

func (m *ItemManager) DeleteCategory(id string) error {
	d := m.storage.Data()
	for i := range d.Categories {
		if d.Categories[i].ID == id {
			d.Categories = append(d.Categories[:i], d.Categories[i+1:]...)
			kept := d.Items[:0]
			for _, it := range d.Items {
				if it.CategoryID != id {
					kept = append(kept, it)
				}
			}
			d.Items = kept
			return m.storage.Save()
		}
	}
	return errors.New("分类不存在")
}

func (m *ItemManager) AddItem(item Item) (Item, error) {
	if item.Name == "" || item.Target == "" {
		return Item{}, errors.New("名称和目标不能为空")
	}
	if item.ID == "" {
		item.ID = GenerateID("item")
	}
	if item.CategoryID == "" {
		if len(m.storage.Data().Categories) == 0 {
			return Item{}, errors.New("请先创建分类")
		}
		item.CategoryID = m.storage.Data().Categories[0].ID
	}
	d := m.storage.Data()
	d.Items = append(d.Items, item)
	return item, m.storage.Save()
}

func (m *ItemManager) UpdateItem(item Item) error {
	d := m.storage.Data()
	for i := range d.Items {
		if d.Items[i].ID == item.ID {
			d.Items[i] = item
			return m.storage.Save()
		}
	}
	return errors.New("项目不存在")
}

func (m *ItemManager) DeleteItem(id string) error {
	d := m.storage.Data()
	for i := range d.Items {
		if d.Items[i].ID == id {
			d.Items = append(d.Items[:i], d.Items[i+1:]...)
			return m.storage.Save()
		}
	}
	return errors.New("项目不存在")
}

func (m *ItemManager) MoveItem(id, toCategoryID string, newSortOrder int) error {
	d := m.storage.Data()
	for i := range d.Items {
		if d.Items[i].ID == id {
			d.Items[i].CategoryID = toCategoryID
			d.Items[i].SortOrder = newSortOrder
			return m.storage.Save()
		}
	}
	return errors.New("项目不存在")
}

// ImportFiles 处理拖放导入：普通文件/快捷方式加入当前分类（或第一个分类）
func (m *ItemManager) ImportFiles(files []string, categoryID string) []Item {
	if len(files) == 0 {
		return nil
	}
	d := m.storage.Data()
	if categoryID == "" && len(d.Categories) > 0 {
		categoryID = d.Categories[0].ID
	}
	added := make([]Item, 0, len(files))
	for _, f := range files {
		item := Item{
			ID:         GenerateID("item"),
			Name:       filepath.Base(f),
			Target:     f,
			CategoryID: categoryID,
		}
		if abs, err := filepath.Abs(f); err == nil {
			item.Target = abs
		}
		if _, err := os.Stat(item.Target); err != nil {
			continue
		}
		if item.CategoryID != "" {
			d.Items = append(d.Items, item)
			added = append(added, item)
		}
	}
	if len(added) > 0 {
		_ = m.storage.Save()
	}
	return added
}

func (m *ItemManager) ItemsByCategory(categoryID string) []Item {
	d := m.storage.Data()
	items := make([]Item, 0)
	for _, it := range d.Items {
		if it.CategoryID == categoryID {
			items = append(items, it)
		}
	}
	sort.Slice(items, func(i, j int) bool { return items[i].SortOrder < items[j].SortOrder })
	return items
}

func (m *ItemManager) IncrementRunCount(id string) {
	d := m.storage.Data()
	for i := range d.Items {
		if d.Items[i].ID == id {
			d.Items[i].RunCount++
			_ = m.storage.Save()
			return
		}
	}
}
```

```go
package app

import (
	"errors"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
)

type Runner struct{}

// ExpandParams 展开内置参数变量：%mp% 当前目录、%mr% 当前盘符
func ExpandParams(target, arguments, workingDir string) string {
	wd := workingDir
	if wd == "" {
		wd, _ = os.Getwd()
	}
	mp := wd
	mr := ""
	if len(wd) >= 2 && wd[1] == ':' {
		mr = wd[:2]
	}
	s := target
	if arguments != "" {
		s += " " + arguments
	}
	s = strings.ReplaceAll(s, "%mp%", mp)
	s = strings.ReplaceAll(s, "%mr%", mr)
	return s
}

// Run 启动项目；runAsAdmin 时以管理员权限运行（Windows ShellExecute runas）
func (r *Runner) Run(item Item) (RunResult, error) {
	if _, err := os.Stat(item.Target); err != nil {
		return RunResult{}, errors.New("目标文件不存在: " + item.Target)
	}
	full := ExpandParams(item.Target, item.Arguments, item.WorkingDir)

	var cmd *exec.Cmd
	var err error
	if item.RunAsAdmin {
		err = runAsAdmin(full, item.WorkingDir)
	} else {
		parts := splitCommand(full)
		cmd = exec.Command(parts[0], parts[1:]...)
		if item.WorkingDir != "" {
			cmd.Dir = item.WorkingDir
		} else {
			cmd.Dir = filepath.Dir(item.Target)
		}
		err = cmd.Start()
	}
	if err != nil {
		return RunResult{}, err
	}
	return RunResult{Success: true}, nil
}

// splitCommand 简单拆分 "target arg1 arg2"；带引号路径用 windows 规则处理
func splitCommand(s string) []string {
	var parts []string
	var cur strings.Builder
	inQuote := false
	for _, c := range s {
		switch c {
		case '"':
			inQuote = !inQuote
		case ' ':
			if inQuote {
				cur.WriteRune(c)
			} else if cur.Len() > 0 {
				parts = append(parts, strings.Trim(cur.String(), `"`))
				cur.Reset()
			}
		default:
			cur.WriteRune(c)
		}
	}
	if cur.Len() > 0 {
		parts = append(parts, strings.Trim(cur.String(), `"`))
	}
	return parts
}
```

```go
//go:build windows

package app

import (
	"errors"
	"strconv"
	"syscall"
	"unsafe"
)

func runAsAdmin(commandLine, workingDir string) error {
	shellExecuteW := syscall.NewLazyDLL("shell32.dll").NewProc("ShellExecuteW")
	verb, _ := syscall.UTF16PtrFromString("runas")
	file, _ := syscall.UTF16PtrFromString(commandLine)
	cwd, _ := syscall.UTF16PtrFromString(workingDir)
	// ShellExecuteW(hwnd=0, verb, file, -parameters, cwd, SW_SHOWNORMAL)
	// 返回 HINSTANCE，值 > 32 表示成功
	ret, _, _ := shellExecuteW.Call(
		0,
		uintptr(unsafe.Pointer(verb)),
		uintptr(unsafe.Pointer(file)),
		0,
		uintptr(unsafe.Pointer(cwd)),
		syscall.SW_SHOWNORMAL,
	)
	if int(ret) <= 32 {
		return errors.New("run as admin failed with code: " + strconv.Itoa(int(ret)))
	}
	return nil
}
```

> 若 `syscall.SW_SHOWNORMAL` 不可用，改用 `1`（SW_SHOWNORMAL 的值）。此函数仅存在于 `admin.go`（build tag windows），不会影响跨平台编译。

> 注意：若 `syscall.ShellExecute` 签名与当前 Go 版本不符，改为通过 `golang.org/x/sys/windows` 加载 `shell32.dll` 的 `ShellExecuteW`（实现细节以编译通过为准）。

```go
func TestSplitCommand(t *testing.T) {
```

```go
package app

import "github.com/wailsapp/wails/v3/pkg/application"

type HotkeyManager struct {
	app      *application.App
	window   *application.WebviewWindow
	windowVisible bool
}

func NewHotkeyManager(a *application.App, w *application.WebviewWindow) *HotkeyManager {
	return &HotkeyManager{app: a, window: w, windowVisible: true}
}

// Register 注册系统级全局快捷键（应用无焦点时也生效）
func (h *HotkeyManager) Register(accelerator string) error {
	return h.app.GlobalShortcut.Register(accelerator, func() {
		if h.windowVisible {
			h.window.Hide()
			h.windowVisible = false
		} else {
			h.window.Show()
			h.window.Focus()
			h.windowVisible = true
		}
	})
}

func (h *HotkeyManager) Unregister(accelerator string) {
	h.app.GlobalShortcut.Unregister(accelerator)
}
```

> API 依据官方示例 `v3/examples/global-shortcuts/main.go`：`app.GlobalShortcut.Register(accel, fn) error`、`app.GlobalShortcut.Unregister(accel)`。若实际签名有出入，以编译为准微调。

```go
package app

import (
	"io/fs"

	"github.com/wailsapp/wails/v3/pkg/application"
	"github.com/wailsapp/wails/v3/pkg/events"
)

type GoRunApp struct {
	App         *application.App
	Window      *application.WebviewWindow
	ItemManager *ItemManager
	Runner      *Runner
	Storage     *Storage
	Hotkey      *HotkeyManager
}

// NewGoRunApp 组装应用。assetsFS 与 trayIcon 由 main.go 通过 //go:embed 注入
// （embed 指令要求所在文件与资源位于同一或子目录，无法从 internal/app 引用上级目录）。
func NewGoRunApp(exeDir string, assets fs.FS, trayIcon []byte) (*GoRunApp, error) {
	storage := NewStorage(exeDir)
	if err := storage.Load(); err != nil {
		return nil, err
	}
	itemManager := NewItemManager(storage)
	runner := &Runner{}

	app := application.New(application.Options{
		Name:        "GoRun",
		Description: "极速启动工具",
		Services: []application.Service{
			application.NewService(itemManager),
			application.NewService(runner),
		},
		Assets: application.AssetOptions{
			Handler: application.BundledAssetFileServer(assets),
		},
	})

	cfg := storage.Data().Config
	window := app.Window.NewWithOptions(application.WebviewWindowOptions{
		Title:          "GoRun",
		Width:          cfg.WindowWidth,
		Height:         cfg.WindowHeight,
		X:              cfg.WindowX,
		Y:              cfg.WindowY,
		EnableFileDrop: true,
	})

	// 拖放导入：事件转发给服务与前端
	window.OnWindowEvent(events.Common.WindowFilesDropped, func(event *application.WindowEvent) {
		files := event.Context().DroppedFiles()
		if len(files) == 0 {
			return
		}
		added := itemManager.ImportFiles(files, "")
		if len(added) > 0 {
			app.Event.Emit("files-imported", map[string]any{"items": added})
		}
	})

	// 窗口位置持久化
	window.OnWindowEvent(events.Common.WindowMoved, func(event *application.WindowEvent) {
		cfg := storage.Data().Config
		x, y := window.Position()
		cfg.WindowX, cfg.WindowY = x, y
		_ = storage.Save()
	})

	hotkey := NewHotkeyManager(app, window)
	if err := hotkey.Register(storage.Data().Config.GlobalHotkey); err != nil {
		return nil, err
	}

	SetupTray(app, window, trayIcon)
	return &GoRunApp{App: app, Window: window, ItemManager: itemManager, Runner: runner, Storage: storage, Hotkey: hotkey}, nil
}
```

```vue
  e.preventDefault()
  contextItem.value = item
  menuX.value = e.clientX

  contextItem.value = null
}

async function removeItem() {
  if (contextItem.value && confirm(`删除「${contextItem.value.name}」？`)) {
    await ItemManager.DeleteItem(contextItem.value.id)
    emit('changed')
  }
  closeMenu()
}

function onDragStart(e: DragEvent, item: Item) {
  draggingId.value = item.id
  e.dataTransfer?.setData('text/plain', item.id)
}

async function onDrop(e: DragEvent, target: Item) {
  e.preventDefault()
  const id = e.dataTransfer?.getData('text/plain') || draggingId.value
  if (!id || id === target.id) return
  const items = [...props.items]
  const from = items.findIndex(i => i.id === id)
  const to = items.findIndex(i => i.id === target.id)
  if (from < 0 || to < 0) return
  const [moved] = items.splice(from, 1)
  items.splice(to, 0, moved)
  for (let i = 0; i < items.length; i++) {
    await ItemManager.MoveItem(items[i].id, props.categoryId, i)
  }
  draggingId.value = ''
  emit('changed')
}
```

### 风险与注意

1. **Wails v3 处于 beta**：API 可能随版本微调，实现时以本仓库实际 `go get` 版本编译为准。
2. **WebView2 运行时**：Windows 10/11 通常预装；无运行时环境需提示安装（官方提供 `WebviewBrowserPath` 固定版运行时选项）。
3. **图标提取**（`icon.go` 未列入本次任务）：MVP 阶段项目卡片先显示名称文本，图标提取作为后续迭代（与搜索功能同期规划）。
4. **管理员运行**：使用 `syscall.NewLazyDLL("shell32.dll").NewProc("ShellExecuteW")` + `runas` verb（任务 5 步骤 2），仅存在于 Windows build tag；若 `syscall.SW_SHOWNORMAL` 常量不可用则用字面量 `1`。
5. **本地 Linux 环境无法运行 WebView2 实机验证**，功能结论依赖 CI 产物人工验证；代码层面以 `go vet` / `go test` / 前端 `vue-tsc` 把关。
6. **不得安装/编译 Rust 相关包**：本计划全部为 Go/前端代码，与 Rust 无关，不触碰 `rs/` 目录。
